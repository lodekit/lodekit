- [ ] **Step 2: Write the derivation check script (scratchpad, not committed)**

Write to `<scratchpad>/check-marks.mts`:

```typescript
// Proves MARK_BODIES === inner markup of the v2 running SVGs (modulo whitespace),
// with #000 → currentColor. Exits 1 on any mismatch.
import { readFileSync } from "node:fs";
import { MARK_BODIES } from "/Users/sesv/Code/worktrees/lodekit/design/packages/engine/src/marks.ts";

const SRC = "/Users/sesv/Desktop/lodekit/menubar-icons v2/svg";

// Normalize both sides identically; only equality matters, not the normal form.
const norm = (s: string) =>
  s
    .replace(/<!--[\s\S]*?-->/g, "")
    .replace(/\s+/g, " ")
    .replace(/> </g, "><")
    .replace(/"\s+/g, '"')
    .replace(/\s+"/g, '"')
    .trim();

let ok = true;
for (const style of ["a", "b", "c"] as const) {
  const svg = readFileSync(`${SRC}/jar-cloud-${style}-running.svg`, "utf8");
  const want = norm(
    svg
      .replace(/^[\s\S]*?<svg[^>]*>/, "")
      .replace(/<\/svg>\s*$/, "")
      .replace(/#000\b/g, "currentColor"),
  );
  const got = norm(MARK_BODIES[style]);
  if (want === got) {
    console.log(`style ${style}: OK`);
  } else {
    ok = false;
    console.error(`style ${style}: MISMATCH\n  want: ${want}\n  got:  ${got}`);
  }
}
process.exit(ok ? 0 : 1);
```
