# Manager v0.1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn the Electron dev shell into the canonical distribution — a packaged macOS menubar app that carries the whole platform (pinned Node + runtime + MCP + skill), supervises the runtime, and needs nothing preinstalled.

**Architecture:** Everything-bundled, run-in-place (spec §3): a Forge-packaged `Lodekit.app` whose `Resources/` holds a pinned Node and a relocatable `pnpm deploy` payload. A plain-CJS `Supervisor` state machine (no Electron imports, fully unit-tested) spawns the runtime as a single direct `node --import tsx` child; `main.js` is thin Electron wiring (tray, menu, login item, first-run). Spec: `docs/specs/2026-07-02-manager-v0.1-design.md`.

**Tech Stack:** Electron 33 + Electron Forge 7 (maker-zip), pnpm 10 workspaces (`pnpm deploy` in legacy mode), vitest, Node 24.12.0 (pinned), tsx.

## Global Constraints

- **Spec:** `docs/specs/2026-07-02-manager-v0.1-design.md` — every requirement below traces to it.
- **Platform:** macOS arm64 only. Ad-hoc code signature (`osxSign: { identity: "-" }`); Developer ID is a dormant toggle.
- **Pinned Node:** `24.12.0`, pinned in exactly one place: `NODE_VERSION` in `manager/scripts/prepare-bundle.mjs`.
- **Stable path contract (spec §8):** `Resources/node/bin/node`, `Resources/platform/runtime/src/index.ts`, `Resources/platform/mcp/index.ts`, `Resources/platform/skill/SKILL.md`. Do not change these without updating README snippets.
- **The runtime is not modified** except moving `tsx` from devDependencies to dependencies. All manager↔runtime contact is env vars (`LODEKIT_ROOT`, `LODEKIT_CACHE_DIR`, `LODEKIT_PORT`) + HTTP.
- **Runtime writes never land inside the bundle** — only the Lodekit root, cache dir, and log dir.
- **Style:** manager source is plain CommonJS JS (matches existing `main.js`); scripts are ESM `.mjs`. Canonical terms per `context.md` (*manager*, *runtime*, *dashboard* — never "desktop app", "server").
- **Menubar icon (plan-time decision per spec §6):** text glyph via `tray.setTitle("⬡")`, `"⬡ !"` in error state, with an empty `nativeImage` — no binary assets at v0.1; real template icons arrive with Developer ID packaging polish.
- **Commits:** on the current `manager` branch, conventional prefixes, **no Co-Authored-By trailers**. Run commands from the repo root unless a step says otherwise.
- Port is 7777; dashboard URL `http://lodekit.localhost:7777`.

```js
const path = require("node:path");

const PORT = 7777;

// The dev/default path-mode switch (spec §7). Everything mode-dependent lives here;
// main.js just spawns `nodeBin --import tsx runtimeEntry` with cwd runtimeDir.
function getPaths({ packaged, homedir, resourcesPath, repoRoot }) {
  const appSupportDir = path.join(homedir, "Library", "Application Support", "Lodekit");
  const common = {
    port: PORT,
    dashboardUrl: `http://lodekit.localhost:${PORT}`,
    appSupportDir,
    pidFile: path.join(appSupportDir, "runtime.pid"),
    logsDir: path.join(homedir, "Library", "Logs", "Lodekit"),
  };
  if (packaged) {
    const platformDir = path.join(resourcesPath, "platform");
    return {
      ...common,
      mode: "default",
      lodekitRoot: path.join(homedir, "Documents", "Lodekit"),
      cacheDir: path.join(homedir, "Library", "Caches", "Lodekit"),
      nodeBin: path.join(resourcesPath, "node", "bin", "node"),
      runtimeDir: path.join(platformDir, "runtime"),
      runtimeEntry: path.join(platformDir, "runtime", "src", "index.ts"),
      mcpEntry: path.join(platformDir, "mcp", "index.ts"),
    };
  }
  return {
    ...common,
    mode: "dev",
    lodekitRoot: path.join(repoRoot, "examples"),
    cacheDir: path.join(repoRoot, ".cache"),
    nodeBin: "node",
    runtimeDir: path.join(repoRoot, "packages", "runtime"),
    runtimeEntry: path.join(repoRoot, "packages", "runtime", "src", "index.ts"),
    mcpEntry: path.join(repoRoot, "packages", "mcp", "index.ts"),
  };
}
```

```js
"use strict";
const fs = require("node:fs");
const path = require("node:path");

// One log file per process per manager launch; the previous launch's file
// survives as `<name>.old` (spec §6).
function openLogStream(logsDir, name) {
  fs.mkdirSync(logsDir, { recursive: true });
  const file = path.join(logsDir, name);
  if (fs.existsSync(file)) fs.renameSync(file, `${file}.old`);
  return fs.createWriteStream(file, { flags: "a" });
}

module.exports = { openLogStream };
```

The core of spec §5: states `stopped → starting → running | error`, single direct child, health polling, stale-PID cleanup, port-conflict detection, SIGTERM→grace→SIGKILL shutdown, no auto-restart. Plain CJS with injected effects so it tests without Electron or real processes.

- Consumes: nothing from other tasks (pidFile path comes from Task 2's `getPaths` at wiring time).

Members: `state` (string), `errorMessage` (string|null), `async start()`, `async stop()`, `async restart()`. Emits `"state"` with `(state, errorMessage)`. Task 4 consumes exactly this surface.

```js
import { EventEmitter } from "node:events";
import fs from "node:fs";
import os from "node:os";
import path from "node:path";
import { Supervisor } from "../src/supervisor.js";

function fakeChild({ obeysSigterm = true } = {}) {
  const child = new EventEmitter();
  child.pid = 4242;
  child.signalsSent = [];
  child.kill = (sig) => {
    child.signalsSent.push(sig);
    if (sig === "SIGTERM" && obeysSigterm) queueMicrotask(() => child.emit("exit", 0, "SIGTERM"));
    if (sig === "SIGKILL") queueMicrotask(() => child.emit("exit", null, "SIGKILL"));
    return true;
  };
  return child;
}

function tmpPidFile() {
  return path.join(fs.mkdtempSync(path.join(os.tmpdir(), "lodekit-sup-")), "runtime.pid");
}

// Small real-time budgets keep the suite fast without fake timers.
const fastOpts = { startTimeoutMs: 500, pollIntervalMs: 10, graceMs: 100 };

describe("Supervisor", () => {

    const child = fakeChild();

    const pidFile = tmpPidFile();
    const sup = new Supervisor({
      ...fastOpts,
      pidFile,
      spawnRuntime: () => ((spawned = true), child),
```

Create `manager/src/supervisor.js`:

```js
"use strict";
const { EventEmitter } = require("node:events");
const fs = require("node:fs");
const path = require("node:path");

const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

// Runtime lifecycle (spec §5): stopped → starting → running | error.
// Effects (spawning, health probes, signals) are injected so this tests
// without Electron or real processes. No auto-restart: failures surface
// as `error` and wait for the person.
class Supervisor extends EventEmitter {
  constructor({
    spawnRuntime,
    checkHealth,
    pidFile,
    killProcess = process.kill,
    startTimeoutMs = 30_000,
    pollIntervalMs = 500,
    graceMs = 5_000,
  }) {
    super();
    this.spawnRuntime = spawnRuntime;
    this.checkHealth = checkHealth;
    this.pidFile = pidFile;
    this.killProcess = killProcess;
    this.startTimeoutMs = startTimeoutMs;
    this.pollIntervalMs = pollIntervalMs;
    this.graceMs = graceMs;
    this.state = "stopped";
    this.errorMessage = null;
    this.child = null;
  }

  setState(state, errorMessage = null) {
    this.state = state;
    this.errorMessage = errorMessage;
    this.emit("state", state, errorMessage);
  }

  processAlive(pid) {
    try {
      this.killProcess(pid, 0);
      return true;
    } catch {
      return false;
    }
  }

  // A pid file left behind by a crashed manager points at a runtime nobody
  // supervises — terminate it before starting fresh (spec §5).
  async cleanupStale() {
    let pid;
    try {
      pid = Number(fs.readFileSync(this.pidFile, "utf8"));
    } catch {
      return;
    }
    if (Number.isInteger(pid) && pid > 0 && this.processAlive(pid)) {
      try {
        this.killProcess(pid, "SIGTERM");
      } catch {}
      const deadline = Date.now() + this.graceMs;
      while (Date.now() < deadline && this.processAlive(pid)) await sleep(25);
      if (this.processAlive(pid)) {
        try {
          this.killProcess(pid, "SIGKILL");
        } catch {}
      }
    }
    fs.rmSync(this.pidFile, { force: true });
  }

  async start() {
    if (this.state === "starting" || this.state === "running") return;
    this.setState("starting");
    await this.cleanupStale();
    if (this.state !== "starting") return; // stopped mid-start
    // Something we didn't record in the pid file already answers on the port
    // (e.g. a dev runtime). Never kill what we didn't start (spec §5).
    if (await this.checkHealth()) {
      this.setState("error", "Port 7777 already in use by another process (a dev runtime?)");
      return;
    }
    let child;
    try {
      child = this.spawnRuntime();
    } catch (e) {
      this.setState("error", `Failed to start runtime: ${e.message}`);
      return;
    }
    this.child = child;
    fs.mkdirSync(path.dirname(this.pidFile), { recursive: true });
    fs.writeFileSync(this.pidFile, String(child.pid));
    child.on("exit", (code, signal) => {
      this.child = null;
      fs.rmSync(this.pidFile, { force: true });
      if (this.state === "running" || this.state === "starting") {
        this.setState("error", `Runtime exited unexpectedly (${signal ?? `code ${code}`})`);
      }
    });
    const deadline = Date.now() + this.startTimeoutMs;
    while (this.state === "starting" && this.child && Date.now() < deadline) {
      if (await this.checkHealth()) {
        if (this.state === "starting" && this.child) this.setState("running");
        return;
      }
      await sleep(this.pollIntervalMs);
    }
    if (this.state === "starting") {
      await this.killChild();
      this.setState(
        "error",
        `Runtime did not become ready within ${this.startTimeoutMs / 1000}s`
      );
    }
  }

  async killChild() {
    if (!this.child) return;
    this.child.kill("SIGTERM");
    const deadline = Date.now() + this.graceMs;
    while (this.child && Date.now() < deadline) await sleep(25);
    if (this.child) {
      this.child.kill("SIGKILL");
      const killDeadline = Date.now() + this.graceMs;
      while (this.child && Date.now() < killDeadline) await sleep(25);
      this.child = null;
    }
  }

  async stop() {
    if (this.state === "stopped") return;
    // Set the state first so the exit handler doesn't report a deliberate
    // shutdown as an unexpected exit.
    this.setState("stopped");
    await this.killChild();
    fs.rmSync(this.pidFile, { force: true });
  }

  async restart() {
    await this.stop();
    await this.start();
  }
}

module.exports = { Supervisor };
```

```js
  child.stderr.pipe(runtimeLogStream, { end: false });

function checkHealth() {
  return new Promise((resolve) => {
    const req = http.get(

      (res) => {
        res.resume();
        resolve(res.statusCode === 200);

    req.on("timeout", () => {
      req.destroy();
      resolve(false);

  const errorMessage = bootError ?? supervisor.errorMessage;

    ? []

      : state === "error"
        ? [{ label: "Restart", click: () => supervisor.restart() }]
        : [{ label: "Start", click: () => supervisor.start() }];
  tray.setContextMenu(
    Menu.buildFromTemplate([
      { label: STATE_LABEL[state], enabled: false },
      ...(state === "error" && errorMessage ? [{ label: errorMessage, enabled: false }] : []),
      { type: "separator" },

        enabled: state === "running",

      },
      { label: "Open Logs", click: () => shell.openPath(paths.logsDir) },
      { type: "separator" },
      { label: "Quit Lodekit", click: quit },

  );

  fs.mkdirSync(path.join(paths.lodekitRoot, "apps"), { recursive: true });
  fs.mkdirSync(path.join(paths.lodekitRoot, "data"), { recursive: true });

  log("quit requested");
  await supervisor.stop();
```

```js
#!/usr/bin/env node
// Builds the packaged manager's bundle inputs (spec §3):
//   build/platform/{runtime,mcp,skill} — relocatable payload (pnpm deploy)
//   build/node                         — pinned Lodekit-local Node (darwin-arm64)
import { execSync } from "node:child_process";
import crypto from "node:crypto";
import fs from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";

// The one place the Lodekit-local Node version is pinned (spec §3).
const NODE_VERSION = "24.12.0";
const NODE_DIST = `node-v${NODE_VERSION}-darwin-arm64`;

const managerDir = path.resolve(path.dirname(fileURLToPath(import.meta.url)), "..");
const repoRoot = path.resolve(managerDir, "..");
const buildDir = path.join(managerDir, "build");
const platformDir = path.join(buildDir, "platform");
const nodeDir = path.join(buildDir, "node");
const cacheDir = path.join(managerDir, ".node-cache");

const run = (cmd, opts = {}) => execSync(cmd, { stdio: "inherit", cwd: repoRoot, ...opts });

// The payload's node_modules are whatever the last `pnpm install` built —
// native modules (better-sqlite3) must match the bundled Node's ABI.
function assertNodeVersion() {
  if (process.version !== `v${NODE_VERSION}`) {
    throw new Error(
      `Bundle with Node v${NODE_VERSION} (found ${process.version}) so native modules ` +
        `match the bundled Node's ABI. Switch Node, then pnpm install (or ` +
        `pnpm rebuild better-sqlite3) and re-run.`
    );
  }
}

async function fetchNode() {
  const marker = path.join(nodeDir, ".lodekit-node-version");
  if (fs.existsSync(marker) && fs.readFileSync(marker, "utf8") === NODE_VERSION) return;
  fs.rmSync(nodeDir, { recursive: true, force: true });
  fs.mkdirSync(cacheDir, { recursive: true });
  const tarball = path.join(cacheDir, `${NODE_DIST}.tar.gz`);
  if (!fs.existsSync(tarball)) {
    console.log(`Downloading ${NODE_DIST}…`);
    const res = await fetch(`https://nodejs.org/dist/v${NODE_VERSION}/${NODE_DIST}.tar.gz`);
    if (!res.ok) throw new Error(`Node download failed: HTTP ${res.status}`);
    fs.writeFileSync(tarball, Buffer.from(await res.arrayBuffer()));
  }
  const shasums = await (
    await fetch(`https://nodejs.org/dist/v${NODE_VERSION}/SHASUMS256.txt`)
  ).text();
  const expected = shasums
    .split("\n")
    .find((l) => l.trim().endsWith(`${NODE_DIST}.tar.gz`))
    ?.split(/\s+/)[0];
  const actual = crypto.createHash("sha256").update(fs.readFileSync(tarball)).digest("hex");
  if (!expected || actual !== expected) {
    fs.rmSync(tarball, { force: true });
    throw new Error("Node tarball checksum mismatch — deleted; re-run to re-download.");
  }
  fs.rmSync(path.join(cacheDir, NODE_DIST), { recursive: true, force: true });
  run(`tar -xzf "${tarball}" -C "${cacheDir}"`);
  fs.mkdirSync(buildDir, { recursive: true });
  fs.renameSync(path.join(cacheDir, NODE_DIST), nodeDir);
  fs.writeFileSync(marker, NODE_VERSION);
}

function buildPayload() {
  fs.rmSync(platformDir, { recursive: true, force: true });
  fs.mkdirSync(platformDir, { recursive: true });
  run(`pnpm --filter @lodekit/runtime --prod deploy "${path.join(platformDir, "runtime")}"`);
  run(`pnpm --filter @lodekit/mcp --prod deploy "${path.join(platformDir, "mcp")}"`);
  fs.cpSync(path.join(repoRoot, "packages", "skill"), path.join(platformDir, "skill"), {
    recursive: true,
    filter: (src) => !src.includes("node_modules"),
  });
}

// Relocatable means no symlink may resolve outside the payload (spec §3);
// internal relative links (node_modules/.bin) are fine.
function assertNoEscapingSymlinks() {
  const links = execSync(`find "${platformDir}" -type l`, { encoding: "utf8" })
    .trim()
    .split("\n")
    .filter(Boolean);
  const root = fs.realpathSync(platformDir) + path.sep;
  for (const link of links) {
    const target = fs.realpathSync(link);
    if (!target.startsWith(root)) {
      throw new Error(`Payload symlink escapes the bundle: ${link} -> ${target}`);
    }
  }
}

// The bundled Node must load the payload's native module — proves the ABI match.
function assertNativeModule() {
  run(`"${path.join(nodeDir, "bin", "node")}" -e "require('better-sqlite3')"`, {
    cwd: path.join(platformDir, "runtime"),
  });
}

assertNodeVersion();
await fetchNode();
buildPayload();
assertNoEscapingSymlinks();
assertNativeModule();
console.log(`Bundle inputs ready: ${buildDir} (Node v${NODE_VERSION} + platform payload)`);
```
