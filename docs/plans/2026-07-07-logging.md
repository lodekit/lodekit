# Platform Logging Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the logging vertical from `docs/specs/2026-07-06-logging-design.md`: structured logs from every platform component and from apps into a dedicated `logs.sqlite`, with retention, a dashboard Logs view, an MCP tool, and the macOS-native paths refactor.

**Architecture:** All records flow through the runtime host process: a zero-dep `logStore.ts` (better-sqlite3 sink) + `logger.ts` (emit facade: insert + SSE broadcast + stderr tee for errors). App output arrives via worker console capture (postMessage) and a new log-store service (SDK `logs` handle). Retention is a scheduler sweep (7 days + 200k rows). The manager's text tier moves from `.old` rotation to daily dated files with a 7-day sweep.

**Tech Stack:** TypeScript (buildless, tsx), Hono (chained routers + `hc`), better-sqlite3, zod v4 + `@hono/zod-validator`, TanStack Start/React (dashboard), vitest. **No new dependencies.**

## Global Constraints

- Spec: `docs/specs/2026-07-06-logging-design.md`. Domain terms: `context.md`.
- Levels: `debug`/`info`/`warn`/`error`, stored numerically as 10/20/30/40. No trace/fatal.
- Retention constants: 7 days (`7 * 24 * 60 * 60 * 1000` ms) and 200_000 rows; manager text logs 7 days.
- Zero new npm dependencies anywhere.
- Chained Hono router style is normative (it powers `hc` type inference) — mirror `kv-store.ts`.
- Catalog name `log-store`, SDK handle `logs`, mounted at `/api/apps/:appId/log-store`.
- Platform records have `app_id NULL`; addressed in query params as `app=platform`.
- Work on the current branch (`stack`), commit per task. Do NOT merge/rebase into `master` — the person integrates. No `Co-Authored-By` trailers.
- Run tests from the repo root with `pnpm --filter <pkg> exec vitest run <file>`; full suite `pnpm test`.
- Logging must never throw into its caller: catch, stderr, continue.

```ts
import fs from "node:fs";
import path from "node:path";
import Database from "better-sqlite3";

export const LEVELS = { debug: 10, info: 20, warn: 30, error: 40 } as const;
export type LevelName = keyof typeof LEVELS;
const NAME_OF: Record<number, LevelName> = { 10: "debug", 20: "info", 30: "warn", 40: "error" };

export interface LogRecord {
  id: number;
  ts: number;
  level: LevelName;
  source: string;
  appId: string | null;
  message: string;
  fields: Record<string, unknown> | null;
}

export interface LogWrite {
  level: LevelName;
  source: string;
  appId?: string | null;
  message: string;
  fields?: Record<string, unknown> | null;
}

export interface LogQuery {
  app?: string; // app id, or "platform" for app_id IS NULL
  level?: LevelName; // minimum
  source?: string;
  before?: number; // id cursor
  limit?: number;
  q?: string; // substring on message
}

const RETENTION_MAX_AGE_MS = 7 * 24 * 60 * 60 * 1000;
const RETENTION_MAX_ROWS = 200_000;

// Single-writer singleton: every record flows through the host process (spec §5).
let db: Database.Database | null = null;

// Opens (creating if needed) <stateDir>/logs.sqlite. Returns false and degrades to
// stderr-only when the store can't open — the runtime must still boot (spec §7).
export function initLogStore(stateDir: string): boolean {
  db?.close();
  db = null;
  try {
    fs.mkdirSync(stateDir, { recursive: true });
    db = new Database(path.join(stateDir, "logs.sqlite"));
    db.pragma("journal_mode = WAL");
    db.exec(
      `CREATE TABLE IF NOT EXISTS logs (
        id      INTEGER PRIMARY KEY,
        ts      INTEGER NOT NULL,
        level   INTEGER NOT NULL,
        source  TEXT NOT NULL,
        app_id  TEXT,
        message TEXT NOT NULL,
        fields  TEXT
      );
      CREATE INDEX IF NOT EXISTS logs_app ON logs(app_id, id);
      CREATE INDEX IF NOT EXISTS logs_level ON logs(level, id);`
    );
    return true;
  } catch (e) {
    db = null;
    process.stderr.write(`[logger] logs.sqlite unavailable — stderr-only logging: ${e}\n`);
    return false;
  }
}

interface Row { id: number; ts: number; level: number; source: string; app_id: string | null; message: string; fields: string | null }

const toRecord = (r: Row): LogRecord => ({
  id: r.id,
  ts: r.ts,
  level: NAME_OF[r.level] ?? "info",
  source: r.source,
  appId: r.app_id,
  message: r.message,
  fields: r.fields ? (JSON.parse(r.fields) as Record<string, unknown>) : null,
});

export function insertLog(w: LogWrite): LogRecord | null {
  if (!db) return null;
  try {
    const ts = Date.now();
    const info = db
      .prepare("INSERT INTO logs (ts, level, source, app_id, message, fields) VALUES (?, ?, ?, ?, ?, ?)")
      .run(ts, LEVELS[w.level], w.source, w.appId ?? null, w.message, w.fields ? JSON.stringify(w.fields) : null);
    return {
      id: Number(info.lastInsertRowid),
      ts,
      level: w.level,
      source: w.source,
      appId: w.appId ?? null,
      message: w.message,
      fields: w.fields ?? null,
    };
  } catch (e) {
    process.stderr.write(`[logger] log write failed: ${e}\n`);
    return null;
  }
}

export function queryLogs(opts: LogQuery = {}): { records: LogRecord[]; nextBefore: number | null } {
  if (!db) return { records: [], nextBefore: null };
  const limit = Math.min(Math.max(1, Math.floor(opts.limit ?? 100)), 1000);
  const where: string[] = [];
  const params: unknown[] = [];
  if (opts.app === "platform") where.push("app_id IS NULL");
  else if (opts.app) { where.push("app_id = ?"); params.push(opts.app); }
  if (opts.level) { where.push("level >= ?"); params.push(LEVELS[opts.level]); }
  if (opts.source) { where.push("source = ?"); params.push(opts.source); }
  if (opts.before != null) { where.push("id < ?"); params.push(opts.before); }
  if (opts.q) {
    where.push("message LIKE ? ESCAPE '\\'");
    params.push("%" + opts.q.replace(/[\\%_]/g, (m) => "\\" + m) + "%");
  }
  const rows = db
    .prepare(
      `SELECT id, ts, level, source, app_id, message, fields FROM logs
       ${where.length ? "WHERE " + where.join(" AND ") : ""}
       ORDER BY id DESC LIMIT ?`
    )
    .all(...params, limit) as Row[];
  const records = rows.map(toRecord);
  return { records, nextBefore: rows.length === limit ? records[records.length - 1].id : null };
}

export function sweepLogStore(opts: { maxAgeMs?: number; maxRows?: number } = {}): number {
  if (!db) return 0;
  const maxAgeMs = opts.maxAgeMs ?? RETENTION_MAX_AGE_MS;
  const maxRows = opts.maxRows ?? RETENTION_MAX_ROWS;
  try {
    let deleted = db.prepare("DELETE FROM logs WHERE ts < ?").run(Date.now() - maxAgeMs).changes;
    // Row at OFFSET maxRows (newest-first) is the first over the cap; NULL subselect (under cap) deletes nothing.
    deleted += db
      .prepare("DELETE FROM logs WHERE id <= (SELECT id FROM logs ORDER BY id DESC LIMIT 1 OFFSET ?)")
      .run(maxRows).changes;
    return deleted;
  } catch (e) {
    process.stderr.write(`[logger] log sweep failed: ${e}\n`);
    return 0;
  }
}

export function closeLogStoreForTest(): void {
  db?.close();
  db = null;
}
```

```ts
import { insertLog, LEVELS, type LevelName } from "./logStore.js";
import { broadcast } from "./sse.js";

export interface Logger {
  debug(message: string, fields?: Record<string, unknown>): void;
  info(message: string, fields?: Record<string, unknown>): void;
  warn(message: string, fields?: Record<string, unknown>): void;
  error(message: string, fields?: Record<string, unknown>): void;
}

// The one emission path (spec §5): persist, stream live, and tee errors to stderr
// so the system stream keeps evidence even when the SQLite sink is the failure.
export function emitLog(
  level: LevelName,
  source: string,
  message: string,
  fields?: Record<string, unknown>,
  appId?: string
): void {
  const record = insertLog({ level, source, appId, message, fields }) ?? {
    id: 0,
    ts: Date.now(),
    level,
    source,
    appId: appId ?? null,
    message,
    fields: fields ?? null,
  };
  broadcast("log", record, appId);
  if (LEVELS[level] >= LEVELS.error) {
    const stack = typeof fields?.stack === "string" ? `\n${fields.stack}` : "";
    process.stderr.write(`[${source}${appId ? `:${appId}` : ""}] ${message}${stack}\n`);
  }
}

export function createLogger(source: string, appId?: string): Logger {
  const at = (level: LevelName) => (message: string, fields?: Record<string, unknown>) =>
    emitLog(level, source, message, fields, appId);
  return { debug: at("debug"), info: at("info"), warn: at("warn"), error: at("error") };
}
```

```ts
import { Hono, type MiddlewareHandler } from "hono";
import { zValidator } from "@hono/zod-validator";
import { z } from "zod";
import { isValidAppId } from "../registry.js";
import { validationHook } from "../validation.js";
import { emitLog } from "../logger.js";
import { queryLogs } from "../logStore.js";

export const CATALOG_NAME = "log-store";

export type LogsEnv = { Variables: { appId: string } };

// Applied by the parent router on /apps/:appId/log-store/* — validates the app id.
export function withLogs(): MiddlewareHandler<LogsEnv> {
  return async (c, next) => {
    const appId = c.req.param("appId");
    if (!appId || !isValidAppId(appId)) return c.json({ error: "invalid app id" }, 400);
    c.set("appId", appId);
    await next();
  };
}

const levelSchema = z.enum(["debug", "info", "warn", "error"]);
const writeBody = z.object({
  level: levelSchema,
  message: z.string().min(1).max(8192),
  fields: z.record(z.string(), z.unknown()).optional(),
});
// Shared by the app-scoped GET and (extended with `app`) the global /api/logs.
export const logsQuery = z.object({
  level: levelSchema.optional(),
  source: z.string().optional(),
  before: z.coerce.number().int().positive().optional(),
  limit: z.coerce.number().optional(),
  q: z.string().optional(),
});

// Chained style is normative — it is what makes hc route-type inference work.
export const logRoutes = new Hono<LogsEnv>()
  .onError((err, c) => c.json({ error: err instanceof Error ? err.message : String(err) }, 400))
  .post("/", zValidator("json", writeBody, validationHook), (c) => {
    const { level, message, fields } = c.req.valid("json");
    emitLog(level, "app", message, fields, c.var.appId);
    return c.json({ ok: true }, 201);
  })
  .get("/", zValidator("query", logsQuery, validationHook), (c) => {
    const { level, source, before, limit, q } = c.req.valid("query");
    return c.json(queryLogs({ app: c.var.appId, level, source, before, limit, q }));
  });

export type LogRoutes = typeof logRoutes;
```

```js
// Console capture (logging spec §5): app console output becomes structured log
// records instead of untagged runtime stdout — keeping the text tier minimal.
// Replaces (not tees) the console methods; the platform log is the destination.
const CONSOLE_LEVELS = { debug: "debug", log: "info", info: "info", warn: "warn", error: "error" };
for (const [method, level] of Object.entries(CONSOLE_LEVELS)) {
  console[method] = (...args) => {
    try {

      const message = args.map((a) => (typeof a === "string" ? a : inspect(a))).join(" ");
      parentPort.postMessage({ type: "log", level, message, fields: err?.stack ? { stack: err.stack } : null });
    } catch {
      // logging must never break the app (e.g. unserializable arg)
    }
  };
}
```

```js
"use strict";
const fs = require("node:fs");
const path = require("node:path");
const { Writable } = require("node:stream");

const DEFAULT_DAYS = 7;

// UTC date stamp — a stable, sortable YYYYMMDD used in filenames and sweep compares.
function dateStamp(d) {
  return d.toISOString().slice(0, 10).replace(/-/g, "");
}

// Deletes dated log files older than `days`, plus the pre-dated-scheme legacy
// names (one-time cleanup; logging spec §7). Lexicographic compare works because
// the stamp is fixed-width YYYYMMDD.
function sweepLogs(logsDir, { days = DEFAULT_DAYS, now = () => new Date() } = {}) {
  const cutoff = dateStamp(new Date(now().getTime() - days * 24 * 60 * 60 * 1000));
  let entries;
  try {
    entries = fs.readdirSync(logsDir);
  } catch {
    return;
  }
  for (const name of entries) {
    const dated = /^[a-z]+-(\d{8})\.log$/.exec(name);
    const legacy = /^(runtime|manager)\.log(\.old)?$/.test(name);
    if ((dated && dated[1] < cutoff) || legacy) {
      try {
        fs.unlinkSync(path.join(logsDir, name));
      } catch {
        // sweep is best-effort; a locked file waits for the next roll
      }
    }
  }
}

// A pipe-compatible Writable appending to <base>-YYYYMMDD.log. Same-day launches
// append (a crash loop must not destroy the launch that started it); the write
// path rolls to a new file when the date stamp changes; sweep runs at open and
// at each roll.
function openDailyLogStream(logsDir, base, { days = DEFAULT_DAYS, now = () => new Date() } = {}) {
  fs.mkdirSync(logsDir, { recursive: true });
  let current = null;
  let stamp = "";
  const roll = () => {
    const s = dateStamp(now());
    if (s !== stamp) {
      if (current) current.end();
      stamp = s;
      sweepLogs(logsDir, { days, now });
      current = fs.createWriteStream(path.join(logsDir, `${base}-${s}.log`), { flags: "a" });
    }
    return current;
  };
  roll();
  return new Writable({
    write(chunk, encoding, callback) {
      roll().write(chunk, encoding, callback);
    },
    final(callback) {
      current ? current.end(callback) : callback();
    },
  });
}

module.exports = { dateStamp, openDailyLogStream, sweepLogs };
```
