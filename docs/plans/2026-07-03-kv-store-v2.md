# kv-store v2 (Storage Iteration) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Spec:** `docs/specs/2026-07-03-kv-store-v2-design.md` — read it before starting any task.

**Goal:** Replace the `db`+`store` storage surface with kv-store v2: six Redis-derived KV ops + TTL + an SSE change feed, served from a new runtime `services/` convention, consumed through an isomorphic SDK `createClient` (`kv` handle, collection sugar) by apps, MCP, and the dashboard — with both example apps migrated off SQL.

**Architecture:** The runtime gains `src/services/kv-store.ts` (schema + ops + Express router mounted at `/api/apps/:slug/kv-store/`) and a tiny internal scheduler for the expiry sweep. SSE gains per-slug scoping (`/api/apps/:slug/events`). The SDK splits into a transport-agnostic `createClient` core (used by MCP and the dashboard with explicit `{baseUrl, slug}`) and an app entry whose slug is injected at build time via Vite `define`. Old `/db/*` routes, the SDK `db` export, and MCP `inspect_db` are retired at the end, once apps have migrated.

**Tech Stack:** Node/Express + better-sqlite3 (runtime, buildless via tsx), Vite JS API (builds), vitest (tests), TanStack Query v5 (dashboard panel only), MCP SDK + zod (MCP server).

```ts
// packages/runtime/src/scheduler.ts
// Runtime-internal recurring tasks (NOT the future person-facing Tasks service).
// Tasks are error-isolated (a throw logs and continues) and never hold the
// process open (unref).

const tasks = new Map<string, { run: () => void; timer: NodeJS.Timeout }>();

export function every(name: string, intervalMs: number, fn: () => void): void {
  stop(name);
  const run = () => {
    try {
      fn();
    } catch (e) {
      console.error(`[scheduler] task "${name}" failed:`, e);
    }
  };
  const timer = setInterval(run, intervalMs);
  timer.unref();
  tasks.set(name, { run, timer });
}

export function tick(name: string): void {
  tasks.get(name)?.run();
}

export function stop(name: string): void {
  const t = tasks.get(name);
  if (t) {
    clearInterval(t.timer);
    tasks.delete(name);
  }
}

export function stopAll(): void {
  for (const name of [...tasks.keys()]) stop(name);
}
```

Pure data layer: table schema with in-place migration of POC-era tables, and the operations. No HTTP yet. `db.ts` is only touched to expose open connections (with slugs) for the sweep.

**Files:**
- Create: `packages/runtime/src/services/kv-store.ts`
- Modify: `packages/runtime/src/db.ts` (connections map carries slug; add `openDbs()`)
- Test: `packages/runtime/tests/kv-store.test.ts`

**Interfaces:**
- Consumes: `getAppDb(dataDir, slug)` from `../db.js`.
- Produces (all exported from `services/kv-store.ts`):
  - `interface Entry { key: string; value: unknown; createdAt: number; updatedAt: number; expiresAt: number | null }`
  - `class KvError extends Error` (validation/semantic errors → HTTP 400 in Task 4)
  - `getKv(dataDir: string, slug: string): Database.Database` — connection + schema ensured (idempotent per connection)
  - `get(db, key): { found: boolean; entry?: Entry }` — expired reads as missing (lazy delete)
  - `set(db, key, value, opts?: { ttl?: number }): void` — upsert; preserves `created_at`; `ttl` seconds; **set without ttl clears expiry**
  - `del(db, key): boolean`
  - `incr(db, key, by?: number): number` — missing starts at 0; non-number value throws `KvError`; preserves an existing TTL (Redis semantics)
  - `list(db, opts?: { prefix?: string; limit?: number; cursor?: string }): { entries: Entry[]; nextCursor: string | null }` — key-ascending, `%`/`_` escaped, expired filtered, limit clamped to [1, 1000] (default 1000), `cursor` = last key of previous page
  - `getMany(db, keys: string[]): Entry[]` — found entries only
  - `setMany(db, entries: { key: string; value: unknown }[], opts?: { ttl?: number }): void` — one transaction
  - `sweep(db): string[]` — deletes expired rows, returns their keys
  - `sweepAll(): { slug: string; keys: string[] }[]` — sweeps every open connection
- From `db.ts`: `openDbs(): { slug: string; db: Database.Database }[]`

```ts
const connections = new Map<string, { slug: string; db: Database.Database }>();

export function getAppDb(dataDir: string, slug: string): Database.Database {
  const appDataDir = path.join(dataDir, slug);
  const dbPath = path.join(appDataDir, "db.sqlite");
  const cached = connections.get(dbPath);
  if (cached) return cached.db;
  fs.mkdirSync(appDataDir, { recursive: true });
  const db = new Database(dbPath);
  db.pragma("journal_mode = WAL");
  db.exec("CREATE TABLE IF NOT EXISTS _lodekit_store (key TEXT PRIMARY KEY, value TEXT)");
  connections.set(dbPath, { slug, db });
  return db;
}

export function openDbs(): { slug: string; db: Database.Database }[] {
  return [...connections.values()];
}
```

```ts
// packages/runtime/src/services/kv-store.ts
import type Database from "better-sqlite3";
import { getAppDb, openDbs } from "../db.js";

export const CATALOG_NAME = "kv-store";

export interface Entry {
  key: string;
  value: unknown;
  createdAt: number;
  updatedAt: number;
  expiresAt: number | null;
}

/** Validation/semantic error — mapped to HTTP 400 by the router. */
export class KvError extends Error {}

interface Row { key: string; value: string; created_at: number; updated_at: number; expires_at: number | null }

const toEntry = (r: Row): Entry => ({
  key: r.key,
  value: JSON.parse(r.value),
  createdAt: r.created_at,
  updatedAt: r.updated_at,
  expiresAt: r.expires_at,
});

const ensured = new WeakSet<Database.Database>();

export function getKv(dataDir: string, slug: string): Database.Database {
  const db = getAppDb(dataDir, slug);
  if (!ensured.has(db)) {
    ensureSchema(db);
    ensured.add(db);
  }
  return db;
}

// Creates the v2 table, or migrates a POC-era (key,value) table in place by
// adding the metadata columns (timestamps backfilled with "now"). Migrated
// columns lack NOT NULL — acceptable; all writers populate them.
export function ensureSchema(db: Database.Database): void {
  db.exec(
    `CREATE TABLE IF NOT EXISTS _lodekit_store (
      key        TEXT PRIMARY KEY,
      value      TEXT NOT NULL,
      created_at INTEGER NOT NULL,
      updated_at INTEGER NOT NULL,
      expires_at INTEGER
    )`
  );
  const cols = new Set(
    (db.prepare("PRAGMA table_info(_lodekit_store)").all() as { name: string }[]).map((c) => c.name)
  );
  const missing = ["created_at", "updated_at", "expires_at"].filter((c) => !cols.has(c));
  for (const col of missing) db.exec(`ALTER TABLE _lodekit_store ADD COLUMN ${col} INTEGER`);
  if (missing.includes("created_at")) {
    db.prepare("UPDATE _lodekit_store SET created_at = ?, updated_at = ? WHERE created_at IS NULL")
      .run(Date.now(), Date.now());
  }
  db.exec(
    "CREATE INDEX IF NOT EXISTS idx_store_expires ON _lodekit_store(expires_at) WHERE expires_at IS NOT NULL"
  );
}

export function get(db: Database.Database, key: string): { found: boolean; entry?: Entry } {
  const row = db
    .prepare("SELECT key, value, created_at, updated_at, expires_at FROM _lodekit_store WHERE key = ?")
    .get(key) as Row | undefined;
  if (!row) return { found: false };
  if (row.expires_at !== null && row.expires_at <= Date.now()) {
    db.prepare("DELETE FROM _lodekit_store WHERE key = ?").run(key);
    return { found: false };
  }
  return { found: true, entry: toEntry(row) };
}

export function set(
  db: Database.Database,
  key: string,
  value: unknown,
  opts: { ttl?: number } = {}
): void {
  const now = Date.now();
  const expiresAt = opts.ttl != null ? now + opts.ttl * 1000 : null;
  db.prepare(
    `INSERT INTO _lodekit_store (key, value, created_at, updated_at, expires_at)
     VALUES (@key, @value, @now, @now, @expiresAt)
     ON CONFLICT(key) DO UPDATE SET
       value = excluded.value, updated_at = excluded.updated_at, expires_at = excluded.expires_at`
  ).run({ key, value: JSON.stringify(value), now, expiresAt });
}

export function del(db: Database.Database, key: string): boolean {
  return db.prepare("DELETE FROM _lodekit_store WHERE key = ?").run(key).changes > 0;
}

// Synchronous transaction = race-free in the single runtime process.
// Preserves an existing TTL (Redis INCR semantics) by not touching expires_at.
export function incr(db: Database.Database, key: string, by = 1): number {
  const txn = db.transaction((): number => {
    const cur = get(db, key);
    let base = 0;
    if (cur.found) {
      if (typeof cur.entry!.value !== "number") throw new KvError(`incr: value at "${key}" is not a number`);
      base = cur.entry!.value;
    }
    const next = base + by;
    const now = Date.now();
    db.prepare(
      `INSERT INTO _lodekit_store (key, value, created_at, updated_at, expires_at)
       VALUES (?, ?, ?, ?, NULL)
       ON CONFLICT(key) DO UPDATE SET value = excluded.value, updated_at = excluded.updated_at`
    ).run(key, JSON.stringify(next), now, now);
    return next;
  });
  return txn();
}

export function list(
  db: Database.Database,
  opts: { prefix?: string; limit?: number; cursor?: string } = {}
): { entries: Entry[]; nextCursor: string | null } {
  const limit = Math.min(Math.max(1, Math.floor(opts.limit ?? 1000)), 1000);
  const like = (opts.prefix ?? "").replace(/[\\%_]/g, (m) => "\\" + m) + "%";
  const rows = db
    .prepare(
      `SELECT key, value, created_at, updated_at, expires_at FROM _lodekit_store
       WHERE key LIKE ? ESCAPE '\\' AND key > ?
         AND (expires_at IS NULL OR expires_at > ?)
       ORDER BY key ASC LIMIT ?`
    )
    .all(like, opts.cursor ?? "", Date.now(), limit) as Row[];
  const entries = rows.map(toEntry);
  return { entries, nextCursor: rows.length === limit ? entries[entries.length - 1].key : null };
}

export function getMany(db: Database.Database, keys: string[]): Entry[] {
  const out: Entry[] = [];
  for (const key of keys) {
    const r = get(db, key);
    if (r.found) out.push(r.entry!);
  }
  return out;
}

export function setMany(
  db: Database.Database,
  entries: { key: string; value: unknown }[],
  opts: { ttl?: number } = {}
): void {
  const txn = db.transaction(() => {
    for (const e of entries) set(db, e.key, e.value, opts);
  });
  txn();
}

export function sweep(db: Database.Database): string[] {
  return (
    db
      .prepare("DELETE FROM _lodekit_store WHERE expires_at IS NOT NULL AND expires_at <= ? RETURNING key")
      .all(Date.now()) as { key: string }[]
  ).map((r) => r.key);
}

// Sweeps every open connection. Apps never opened since boot aren't swept
// until first access — lazy expiry on read covers correctness; the sweep is
// for space. Only connections opened through getKv have the v2 schema
// guaranteed, so guard with ensureSchema idempotently.
export function sweepAll(): { slug: string; keys: string[] }[] {
  const results: { slug: string; keys: string[] }[] = [];
  for (const { slug, db } of openDbs()) {
    if (!ensured.has(db)) {
      ensureSchema(db);
      ensured.add(db);
    }
    const keys = sweep(db);
    if (keys.length > 0) results.push({ slug, keys });
  }
  return results;
}
```

```ts
// packages/runtime/src/sse.ts
import type { Response } from "express";

interface Client { res: Response; slug?: string }

const clients = new Set<Client>();

// A client with a slug receives only events broadcast for that slug.
// A client without a slug (the global /api/events channel) receives everything.
export function addClient(res: Response, slug?: string): void {
  const client: Client = { res, slug };
  clients.add(client);
  res.on("close", () => clients.delete(client));
}

export function broadcast(event: string, data: unknown, slug?: string): void {
  const payload = `event: ${event}\ndata: ${JSON.stringify(data)}\n\n`;
  for (const c of clients) {
    if (c.slug === undefined || c.slug === slug) c.res.write(payload);
  }
}
```

```ts
import express, { type Request, type Response, type NextFunction } from "express";
import type { Config } from "../config.js";
import { isValidSlug } from "../registry.js";
import { broadcast } from "../sse.js";

const EVENT = `${CATALOG_NAME}-change`;

const validKey = (k: unknown): k is string =>
  typeof k === "string" && k.length > 0 && k.length <= 512;

function emit(slug: string, key: string, op: "set" | "del" | "expire"): void {
  broadcast(EVENT, { slug, key, op }, slug);
}

export function sweepAllAndBroadcast(): void {
  for (const { slug, keys } of sweepAll()) {
    for (const key of keys) emit(slug, key, "expire");
  }
}

interface KvRequest extends Request {
  kvDb?: Database.Database;
}

// Router mounted at /apps/:slug/kv-store (mergeParams gives it :slug).
export function kvRouter(cfg: Config): express.Router {
  const r = express.Router({ mergeParams: true });

  r.use((req: KvRequest, res: Response, next: NextFunction) => {
    const slug = (req.params as { slug: string }).slug;
    if (!isValidSlug(slug)) {
      res.status(400).json({ error: "invalid slug" });
      return;
    }
    req.kvDb = getKv(cfg.dataDir, slug);
    next();
  });

  const handle = (res: Response, fn: () => void) => {
    try {
      fn();
    } catch (e) {
      res.status(400).json({ error: e instanceof Error ? e.message : String(e) });
    }
  };
  const ttlOk = (ttl: unknown) =>
    ttl === undefined || (typeof ttl === "number" && Number.isFinite(ttl) && ttl > 0);
  const slugOf = (req: Request) => (req.params as { slug: string }).slug;

  r.get("/", (req: KvRequest, res) =>
    handle(res, () => {
      const { prefix, cursor } = req.query as { prefix?: string; cursor?: string };
      const limit = req.query.limit !== undefined ? Number(req.query.limit) : undefined;
      res.json(list(req.kvDb!, { prefix, limit, cursor }));
    })
  );

  r.post("/incr", (req: KvRequest, res) =>
    handle(res, () => {
      const { key, by } = req.body ?? {};
      if (!validKey(key) || (by !== undefined && !(typeof by === "number" && Number.isFinite(by)))) {
        res.status(400).json({ error: "invalid key or by" });
        return;
      }
      const value = incr(req.kvDb!, key, by);
      emit(slugOf(req), key, "set");
      res.json({ value });
    })
  );

  r.post("/get-many", (req: KvRequest, res) =>
    handle(res, () => {
      const { keys } = req.body ?? {};
      if (!Array.isArray(keys) || !keys.every(validKey)) {
        res.status(400).json({ error: "keys must be an array of valid keys" });
        return;
      }
      res.json({ entries: getMany(req.kvDb!, keys) });
    })
  );

  r.post("/set-many", (req: KvRequest, res) =>
    handle(res, () => {
      const { entries, ttl } = req.body ?? {};
      const ok =
        Array.isArray(entries) &&
        entries.every((e) => e && typeof e === "object" && validKey(e.key) && "value" in e) &&
        ttlOk(ttl);
      if (!ok) {
        res.status(400).json({ error: "invalid entries or ttl" });
        return;
      }
      setMany(req.kvDb!, entries, { ttl });
      for (const e of entries) emit(slugOf(req), e.key, "set");
      res.json({ ok: true });
    })
  );

  r.get("/:key", (req: KvRequest, res) =>
    handle(res, () => {
      if (!validKey(req.params.key)) {
        res.status(400).json({ error: "invalid key" });
        return;
      }
      const r2 = get(req.kvDb!, req.params.key);
      if (!r2.found) {
        res.json({ found: false });
        return;
      }
      const { value, createdAt, updatedAt, expiresAt } = r2.entry!;
      res.json({ found: true, value, createdAt, updatedAt, expiresAt });
    })
  );

  r.put("/:key", (req: KvRequest, res) =>
    handle(res, () => {
      const body = req.body ?? {};
      if (!validKey(req.params.key) || !("value" in body) || !ttlOk(body.ttl)) {
        res.status(400).json({ error: "invalid key, missing value, or invalid ttl" });
        return;
      }
      set(req.kvDb!, req.params.key, body.value, { ttl: body.ttl });
      emit(slugOf(req), req.params.key, "set");
      res.json({ ok: true });
    })
  );

  r.delete("/:key", (req: KvRequest, res) =>
    handle(res, () => {
      if (!validKey(req.params.key)) {
        res.status(400).json({ error: "invalid key" });
        return;
      }
      const deleted = del(req.kvDb!, req.params.key);
      if (deleted) emit(slugOf(req), req.params.key, "del");
      res.json({ deleted });
    })
  );

  return r;
}
```

```ts
// packages/sdk/src/client.ts
export interface KvEntry<T = unknown> {
  key: string;
  value: T;
  createdAt: number;
  updatedAt: number;
  expiresAt: number | null;
}

export type KvChange = { key: string; op: "set" | "del" | "expire" };

export interface Collection<T> {
  add(value: T): Promise<{ id: number; value: T }>;
  all(): Promise<{ id: number; value: T }[]>;
  get(id: number): Promise<T | undefined>;
  set(id: number, value: T): Promise<void>;
  remove(id: number): Promise<boolean>;
}

export interface Kv {
  get<T = unknown>(key: string): Promise<T | undefined>;
  set(key: string, value: unknown, opts?: { ttl?: number }): Promise<void>;
  del(key: string): Promise<boolean>;
  has(key: string): Promise<boolean>;
  incr(key: string, by?: number): Promise<number>;
  list<T = unknown>(opts?: {
    prefix?: string;
    limit?: number;
    cursor?: string;
  }): Promise<{ entries: KvEntry<T>[]; nextCursor: string | null }>;
  getMany<T = unknown>(keys: string[]): Promise<KvEntry<T>[]>;
  setMany(entries: { key: string; value: unknown }[], opts?: { ttl?: number }): Promise<void>;
  collection<T>(name: string): Collection<T>;
  watch(prefix: string, cb: (change: KvChange) => void): () => void;
}

export interface LodekitClient {
  kv: Kv;
}

async function request<T>(url: string, init?: RequestInit): Promise<T> {
  const res = await fetch(url, init);
  if (!res.ok) {
    let message = res.statusText;
    try {
      const data = (await res.json()) as { error?: string };
      if (data?.error) message = data.error;
    } catch {
      // non-JSON error body — keep statusText
    }
    throw new Error(message);
  }
  return (await res.json()) as T;
}

const jsonInit = (method: string, body: unknown): RequestInit => ({
  method,
  headers: { "Content-Type": "application/json" },
  body: JSON.stringify(body),
});

export function createClient({ baseUrl, slug }: { baseUrl: string; slug: string }): LodekitClient {
  const api = `${baseUrl}/api/apps/${slug}/kv-store`;
  const keyUrl = (key: string) => `${api}/${encodeURIComponent(key)}`;

  type GetResponse<T> = { found: false } | { found: true; value: T };

  // One EventSource per client, shared by all watchers, closed with the last.
  const watchers = new Set<{ prefix: string; cb: (c: KvChange) => void }>();
  let es: { close(): void } | null = null;

  const kv: Kv = {
    async get<T>(key: string): Promise<T | undefined> {
      const r = await request<GetResponse<T>>(keyUrl(key));
      return r.found ? r.value : undefined;
    },
    async set(key, value, opts = {}) {
      await request(keyUrl(key), jsonInit("PUT", { value, ...(opts.ttl != null ? { ttl: opts.ttl } : {}) }));
    },
    async del(key) {
      const r = await request<{ deleted: boolean }>(keyUrl(key), { method: "DELETE" });
      return r.deleted;
    },
    async has(key) {
      return (await request<{ found: boolean }>(keyUrl(key))).found;
    },
    async incr(key, by) {
      const r = await request<{ value: number }>(
        `${api}/incr`,
        jsonInit("POST", by === undefined ? { key } : { key, by })
      );
      return r.value;
    },
    async list(opts = {}) {
      const q = new URLSearchParams();
      if (opts.prefix !== undefined) q.set("prefix", opts.prefix);
      if (opts.limit !== undefined) q.set("limit", String(opts.limit));
      if (opts.cursor !== undefined) q.set("cursor", opts.cursor);
      const qs = q.toString();
      return request(`${api}${qs ? `?${qs}` : ""}`);
    },
    async getMany(keys) {
      const r = await request<{ entries: KvEntry<never>[] }>(`${api}/get-many`, jsonInit("POST", { keys }));
      return r.entries as never;
    },
    async setMany(entries, opts = {}) {
      await request(`${api}/set-many`, jsonInit("POST", { entries, ...(opts.ttl != null ? { ttl: opts.ttl } : {}) }));
    },
    collection<T>(name: string): Collection<T> {
      const itemKey = (id: number) => `${name}:${id}`;
      return {
        async add(value) {
          const id = await kv.incr(`_seq:${name}`);
          await kv.set(itemKey(id), value);
          return { id, value };
        },
        async all() {
          const items: { id: number; value: T; createdAt: number }[] = [];
          let cursor: string | undefined;
          do {
            const page = await kv.list<T>({ prefix: `${name}:`, cursor });
            for (const e of page.entries) {
              items.push({ id: Number(e.key.slice(name.length + 1)), value: e.value, createdAt: e.createdAt });
            }
            cursor = page.nextCursor ?? undefined;
          } while (cursor !== undefined);
          items.sort((a, b) => a.createdAt - b.createdAt);
          return items.map(({ id, value }) => ({ id, value }));
        },
        get: (id) => kv.get<T>(itemKey(id)),
        set: (id, value) => kv.set(itemKey(id), value),
        remove: (id) => kv.del(itemKey(id)),
      };
    },
    watch(prefix, cb) {
      if (typeof EventSource === "undefined") {
        throw new Error("kv.watch requires EventSource (browser only)");
      }
      const watcher = { prefix, cb };
      if (watchers.size === 0) {
        const source = new EventSource(`${baseUrl}/api/apps/${slug}/events`);
        source.addEventListener("kv-store-change", (e) => {
          const change = JSON.parse((e as MessageEvent).data) as KvChange;
          for (const w of watchers) {
            if (change.key.startsWith(w.prefix)) w.cb(change);
          }
        });
        es = source;
      }
      watchers.add(watcher);
      return () => {
        watchers.delete(watcher);
        if (watchers.size === 0 && es) {
          es.close();
          es = null;
        }
      };
    },
  };

  return { kv };
}
```

```tsx
    queryKey: ["kv", slug],
    queryFn: async ({ pageParam }): Promise<KvPage> =>
      (await fetch(
        `/api/apps/${slug}/kv-store?limit=100${pageParam ? `&cursor=${encodeURIComponent(pageParam)}` : ""}`

  });

  return (
    <div className="mt-2 space-y-1 border-t pt-2 text-sm">
      {q.isLoading ? (
        <Skeleton className="h-4" />
      ) : entries.length === 0 ? (

          </div>

      )}
      {q.hasNextPage && (
```
